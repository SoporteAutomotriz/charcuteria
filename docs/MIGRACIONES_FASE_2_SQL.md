# Migraciones SQL de Fase 2 — Base de datos, roles y RLS

Aplica estos 4 bloques en el **SQL Editor** de tu panel de Supabase, **en orden** (1 → 2 → 3 → 4). Cada bloque usa `create table if not exists` y `drop policy if exists`, por lo que son idempotentes.

## Migración 1: Esquema público base

```sql
-- Migración 1: Esquema público base (Fase 2)
-- Tablas: usuarios_personal, categorias, productos, paquetes, paquete_componentes,
--         zonas_transporte, tarifas_montaje, perfiles_consumo, parametros_globales

-- 1. Usuarios del personal (Supabase Auth + rol propio)
create table if not exists public.usuarios_personal (
  id uuid references auth.users not null primary key,
  email text not null,
  rol text not null check (rol in ('propietario', 'administrador')),
  nombre text not null,
  activo boolean not null default true,
  creado_en timestamptz not null default now()
);

-- 2. Categorías del catálogo (sección 4)
create table if not exists public.categorias (
  id bigint generated by default as identity primary key,
  nombre text not null,
  orden_visualizacion int not null default 0,
  activo boolean not null default true,
  creado_en timestamptz not null default now()
);

-- 3. Productos del catálogo (sección 4)
create table if not exists public.productos (
  id bigint generated by default as identity primary key,
  categoria_id bigint references public.categorias(id),
  nombre text not null,
  descripcion text,
  foto_url text,
  orden_visualizacion int not null default 0,
  unidad_compra text not null,
  unidad_venta text not null,
  factor_conversion numeric(10,4) not null default 1,
  merma_pct numeric(5,2) not null default 0,
  porcion_referencia_persona numeric(10,2),
  alergenos text[],
  etiquetas text[],
  estado text not null check (estado in ('activo', 'agotado_temporal', 'inactivo')) default 'activo',
  creado_en timestamptz not null default now(),
  actualizado_en timestamptz not null default now()
);

-- 4. Paquetes (sección 5)
create table if not exists public.paquetes (
  id bigint generated by default as identity primary key,
  nombre text not null,
  foto_url text,
  descripcion text,
  personas_recomendadas int not null default 1,
  creado_en timestamptz not null default now(),
  actualizado_en timestamptz not null default now()
);

-- 5. Componentes de paquetes (sección 5)
create table if not exists public.paquete_componentes (
  id bigint generated by default as identity primary key,
  paquete_id bigint references public.paquetes(id) on delete cascade,
  categoria_id bigint references public.categorias(id),
  nombre text not null,
  cantidad_referencia numeric(10,4) not null default 1,
  tipo text not null check (tipo in ('fijo', 'sustituible', 'opcional', 'quitable')) default 'fijo',
  creado_en timestamptz not null default now()
);

-- 6. Zonas de transporte (sección 7)
create table if not exists public.zonas_transporte (
  id bigint generated by default as identity primary key,
  nombre text not null,
  municipios text[] not null default '{}',

  cargo_distancia numeric(12,2) not null default 0,
  cargo_horario numeric(12,2) not null default 0,
  cargo_acceso_dificil numeric(12,2) not null default 0,
  zona_especial boolean not null default false,
  activo boolean not null default true
);

-- 7. Tarifas de montaje y personal (sección 7)
create table if not exists public.tarifas_montaje (
  id bigint generated by default as identity primary key,
  tarifa_hora numeric(12,2) not null default 0,
  horas_minimas int not null default 0,
  regla_personal_por_invitados int not null default 0,
  activo boolean not null default true
);

-- 8. Perfiles de consumo (sección 7)
create table if not exists public.perfiles_consumo (
  id bigint generated by default as identity primary key,
  tipo_evento text not null,
  duracion_rango text,
  categoria text not null,
  gramos_por_persona numeric(10,2) not null default 0,
  es_comida_principal boolean not null default false,
  activo boolean not null default true
);

-- 9. Parámetros globales (solo propietario puede escribir)
create table if not exists public.parametros_globales (
  id bigint generated by default as identity primary key,
  clave text not null unique,
  valor jsonb not null default '{}',
  descripcion text,
  creado_en timestamptz not null default now(),
  actualizado_en timestamptz not null default now()
);

-- 10. Bitácora de auditoría (solo lectura para todos, escrita por triggers/servidor)
create table if not exists public.bitacora_auditoria (
  id bigint generated by default as identity primary key,
  tabla_afectada text not null,
  registro_id bigint not null,
  accion text not null check (accion in ('insert', 'update', 'delete')),
  valores_anteriores jsonb,
  valores_nuevos jsonb,
  usuario_id uuid,
  ip text,
  fecha timestamptz not null default now()

-- Zonas transporte: propietario y administrador leen/escriben
```

## Migración 2: Cotizaciones, costos privados y versiones

```sql
-- Migración 2: Cotizaciones, versiones y costos privados (Fase 2)
-- Tablas: cotizaciones (vigente), cotizacion_versiones (historial de solo lectura),
--         cotizaciones_costos (privada, solo Propietario, en paralelo a cotizaciones)
-- No hay RLS para el rol anon en ninguna de estas tres tablas: el acceso
-- del público se hace por endpoint de servidor (service_role) y por token largo.
--
-- Modelo de versionado (opción b): cotizaciones guarda la versión vigente.
-- Cuando el estado llega a "enviada" por primera vez, se congela automáticamente
-- como numero_version 1 en cotizacion_versiones (ver trigger en migración 3).

-- 1. Tabla cotizaciones — versión vigente (la que el cliente ve por su enlace privado)
create table if not exists public.cotizaciones (
  id bigint generated by default as identity primary key,
  folio text not null unique check (folio ~ '^CS-\d{4}-\d{4}$'),
  datos_cliente jsonb not null default '{}',
  datos_evento jsonb not null default '{}',
  datos_productos jsonb not null default '{}',
  datos_paquetes jsonb not null default '{}',
  estado text not null check (
    estado in (
      'recibida', 'en_revision', 'enviada', 'aceptada',
      'confirmada', 'realizada', 'cancelada', 'vencida'
    )
  ) default 'recibida',
  vigencia_en timestamptz,
  token text not null unique,
  token_expires_en timestamptz not null,
  total_venta numeric(12,2) not null default 0,
  descuento_pct numeric(5,2) not null default 0 check (descuento_pct between 0 and 100),
  creado_en timestamptz not null default now(),
  actualizado_en timestamptz not null default now(),
  -- Columnas de consentimiento y aviso de privacidad (sección 10)
  fecha_aceptacion_privacidad timestamptz,
  version_aviso_privacidad text,
  canal_contacto_consentido text check (
    canal_contacto_consentido in ('correo', 'whatsapp', 'ambos')
  ),
  creado_por uuid references public.usuarios_personal(id)
);

-- 2. Tabla cotizaciones_costos — privada, solo Propietario, en paralelo a cotizaciones
create table if not exists public.cotizaciones_costos (
  id bigint generated by default as identity primary key,
  cotizacion_id bigint references public.cotizaciones(id) on delete cascade,
  total_costo numeric(12,2) not null default 0,
  datos_costo jsonb not null default '{}',
  creado_en timestamptz not null default now(),
  actualizado_en timestamptz not null default now(),
  unique (cotizacion_id)
);

-- 3. Tabla cotizacion_versiones — historial de solo lectura, una fila por versión
create table if not exists public.cotizacion_versiones (
  id bigint generated by default as identity primary key,
  cotizacion_id bigint references public.cotizaciones(id) on delete cascade,
  numero_version int not null,
  estado text not null,
  total_venta numeric(12,2) not null default 0,
  total_costo numeric(12,2) not null default 0,
  datos_cliente jsonb not null default '{}',
  datos_evento jsonb not null default '{}',
  datos_productos jsonb not null default '{}',
  datos_paquetes jsonb not null default '{}',
  datos_costo_congelados jsonb not null default '{}',
  token text not null,
  fecha_aceptacion_privacidad timestamptz,
  version_aviso_privacidad text,
  canal_contacto_consentido text,
  creado_por uuid references public.usuarios_personal(id),
  creado_en timestamptz not null default now(),
  unique (cotizacion_id, numero_version)
);

-- Índices para búsquedas frecuentes
create index if not exists idx_cotizaciones_token on public.cotizaciones(token);
create index if not exists idx_cotizaciones_estado on public.cotizaciones(estado);
create index if not exists idx_cotizaciones_creado_en on public.cotizaciones(creado_en);
create index if not exists idx_cotizacion_versiones_cotizacion on public.cotizacion_versiones(cotizacion_id);

-- ============================================================
-- POLÍTICAS RLS — Cotizaciones, versiones y costos
-- ============================================================

-- cotizaciones: propietario y administrador leen/escriben todo (excepto costos, que están en tabla aparte)
create policy "cotizaciones: propietario y administrador"
  on public.cotizaciones
  for all
  to authenticated
  using (
    exists (
      select 1 from public.usuarios_personal up
      where up.id = auth.uid() and up.rol in ('propietario', 'administrador')
    )
  );

-- cotizaciones_costos: solo propietario
create policy "cotizaciones_costos: solo propietario"
  on public.cotizaciones_costos
  for all
  to authenticated
  using (
    exists (
      select 1 from public.usuarios_personal up
      where up.id = auth.uid() and up.rol = 'propietario'
    )
  );

-- cotizacion_versiones: solo lectura para propietario y administrador
create policy "cotizacion_versiones: solo lectura propietario y administrador"
  on public.cotizacion_versiones
  for select
  to authenticated
  using (
    exists (
      select 1 from public.usuarios_personal up
      where up.id = auth.uid() and up.rol in ('propietario', 'administrador')
    )
  );

-- Habilitar RLS en las tres tablas
alter table public.cotizaciones enable row level security;
alter table public.cotizaciones_costos enable row level security;
alter table public.cotizacion_versiones enable row level security;
```

## Migración 3: Trigger de congelado automático

```sql
-- Migración 3: Trigger de congelado automático de versiones (Fase 2)
-- Cuando una cotización pasa a estado "enviada" por primera vez, se congela
-- automáticamente como numero_version 1 en cotizacion_versiones, antes de
-- cualquier edición posterior.

-- Función que congela la versión vigente al llegar a "enviada" por primera vez
create or replace function public.fn_congelar_version_al_enviar()
returns trigger
language plpgsql
as $$
begin
  -- Solo actúa cuando el estado nuevo es 'enviada'
  if new.estado <> 'enviada' then
    return new;
  end if;

  -- Solo si aún no existe ninguna versión congelada para esta cotización
  if exists (
    select 1 from public.cotizacion_versiones
    where cotizacion_id = new.id
  ) then
    return new;
  end if;

  -- Congela los datos de la cotización vigente como versión 1
  insert into public.cotizacion_versiones (
    cotizacion_id,
    numero_version,
    estado,
    total_venta,
    total_costo,
    datos_cliente,
    datos_evento,
    datos_productos,
    datos_paquetes,
    datos_costo_congelados,
    token,
    fecha_aceptacion_privacidad,
    version_aviso_privacidad,
    canal_contacto_consentido,
    creado_por,
    creado_en
  )
  select
    new.id,
    1,
    new.estado,
    new.total_venta,
    coalesce(cc.total_costo, 0),
    new.datos_cliente,
    new.datos_evento,
    new.datos_productos,
    new.datos_paquetes,
    cc.datos_costo,
    new.token,
    new.fecha_aceptacion_privacidad,
    new.version_aviso_privacidad,
    new.canal_contacto_consentido,
    new.creado_por,
    now()
  from public.cotizaciones c
  left join public.cotizaciones_costos cc on cc.cotizacion_id = c.id
  where c.id = new.id;

  return new;
end;
$$;

-- Trigger: se dispara antes de actualizar una cotización
create trigger trg_congelar_version_al_enviar
before update of estado on public.cotizaciones
for each row
when (old.estado is distinct from new.estado)
execute function public.fn_congelar_version_al_enviar();
```

## Migración 4: Eliminar política `anon` de tabla `testing` (opcional)

```sql
-- Migración 4: Eliminar política de lectura del rol anon en tabla testing (Fase 2)
-- Seguridad: el rol anónimo no debe leer ninguna tabla directamente.
-- La tabla testing solo se usa desde /api/supabase-test con el cliente de servidor
-- (service_role), por lo que no necesita la política anon.

-- Eliminar la política que concedía SELECT sin restricción al rol anon
drop policy if exists "Permitir SELECT al rol anon en testing" on public.testing;
```
